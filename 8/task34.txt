import re


class Product:
    """Класс товара"""
    def __init__(self, name, price):
        self.name = name
        self.price = price

    def __str__(self):
        return f"{self.name} — {self.price} руб."

    def __repr__(self):
        return f"Product('{self.name}', {self.price})"


class Order:
    """Класс заказа, содержащий список товаров"""
    def __init__(self):
        self._products = []
        self._load_products_from_file()

    def _load_products_from_file(self):
        """Загружает товары из файла list.txt в формате 'название — цена'"""
        try:
            with open('list.txt', 'r', encoding='utf-8') as file:
                for line in file:
                    line = line.strip()
                    if not line:
                        continue

                    # Разделяем строку по тире (разные виды тире: —, –, -)
                    parts = re.split(r'\s*[—–-]\s*', line, maxsplit=1)

                    if len(parts) == 2:
                        name = parts[0].strip()

                        # Извлекаем все числа из части с ценой
                        # Поддерживает форматы: "100", "100.50", "1 200,50"
                        numbers = re.findall(r'\d+(?:[.,]\d+)?', parts[1].replace(' ', ''))

                        if numbers:
                            # Берем первое найденное число как цену
                            price_str = numbers[0].replace(',', '.')
                            try:
                                price = float(price_str)
                                self._products.append(Product(name, price))
                            except ValueError:
                                print(f"Не удалось преобразовать цену в строке: {line}")
                        else:
                            print(f"Число не найдено в строке: {line}")
                    else:
                        print(f"Неверный формат строки: {line}")

        except FileNotFoundError:
            print("Файл list.txt не найден. Создан пустой заказ.")
        except Exception as e:
            print(f"Ошибка при чтении файла: {e}")

    @property
    def total_price(self):
        """Вычисляемое свойство - общая стоимость заказа"""
        return sum(product.price for product in self._products)

    def add_product(self, product):
        """Добавляет товар в заказ"""
        if isinstance(product, Product):
            self._products.append(product)
            print(f"Товар '{product.name}' добавлен в заказ.")
        else:
            print("Ошибка: можно добавлять только объекты класса Product")

    def remove_product(self, product):
        """Удаляет товар из заказа"""
        if product in self._products:
            self._products.remove(product)
            print(f"Товар '{product.name}' удален из заказа.")
        else:
            print(f"Товар '{product.name}' не найден в заказе.")

    def get_products(self):
        """Возвращает список товаров в заказе"""
        return self._products.copy()

    def __str__(self):
        if not self._products:
            return "Заказ пуст"

        result = "Заказ содержит:\n"
        for i, product in enumerate(self._products, 1):
            result += f"{i}. {product}\n"
        result += f"Общая стоимость: {self.total_price:.2f} руб."
        return result
