class User:
    """
    Класс пользователя с валидацией email и хранением хеша пароля.
    """
    
    def __init__(self, email, password):
        # Используем сеттеры для проверки и сохранения значений
        self.email = email
        self.password = password
    
    @property
    def email(self):
        """Геттер для email."""
        return self._email
    
    @email.setter
    def email(self, value):
        """Сеттер для email с простой валидацией на наличие символа @."""
        if not isinstance(value, str) or '@' not in value:
            raise ValueError("Email должен содержать символ '@'")
        self._email = value
    
    @property
    def password(self):
        """
        Геттер для пароля.
        Не возвращает сам пароль — только для демонстрации, что прямой доступ невозможен.
        """
        raise AttributeError("Доступ к паролю запрещён")
    
    @password.setter
    def password(self, value):
        """Сеттер для пароля — сохраняем только его хеш."""
        if not isinstance(value, str) or not value:
            raise ValueError("Пароль должен быть непустой строкой")
        # Храним хеш пароля, а не сам пароль
        self._password_hash = hash(value)
    
    def check_password(self, password):
        """
        Проверяет, соответствует ли хеш переданного пароля сохранённому хешу.
        
        :param password: строка с паролем для проверки
        :return: True, если пароль верный, иначе False
        """
        return hash(password) == self._password_hash
