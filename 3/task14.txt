#одинаковыми значениями в массиве и вывести их индексы.
# Одинаковых значение может быть два и более !
Пример:
mass = [1,2,17,54,30,89,2,1,6,2]


Для числа 1 минимальное расстояние в массиве по индексам: 0 и 7
Для числа 2 минимальное расстояние в массиве по индексам: 6 и 9
Для числа 17 нет минимального расстояния т.к элемент в массиве один.


mass = [3, 7, 11, 44, 10, 1, 7, 6, 3, 44]
positions = {}

for index, value in enumerate(mass):
    if value not in positions:
        positions[value] = []
    positions[value].append(index)
    
for value, indices in positions.items():
    if len (indices) >= 2:
        min_dist = float('inf')
        best_pair = ()
    
        for i in range(len(indices) - 1):
            dist = indices[i+1] - indices[i]
            if dist < min_dist:
                min_dist = dist
                best_pair = (indices[i], indices[i+1])
            print(f"Для числа {value} минимальное расстояние в массиве по индексам: {best_pair[0]} и {best_pair[1]}")
    else:
        print(f"Для числа {value} нет минимального расстояния т.к. элемент в массиве один.")
