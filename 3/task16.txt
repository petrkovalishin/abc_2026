users = [
    {'login': 'Piter', 'age': 23, 'group': "admin"},
    {'login': 'Ivan',  'age': 10, 'group': "guest"},
    {'login': 'Dasha', 'age': 30, 'group': "master"},
    {'login': 'Fedor', 'age': 13, 'group': "guest"},
    {'login': 'Maria', 'age': 26, 'group': "user"},
    {'login': 'Viacheslav', 'age': 24, 'group': "army"},
]

print("Вводим тип сортировки:")
print("1. по возрасту")
print("2. по первой букве логина")
print("3. по группе")

vbr_sort = input("тип сортировки: ")
if vbr_sort == '1':
    crit = int(input("Введите критерии поиска (минимальный возраст): "))
    filter_users = [user for user in users if user['age'] > crit]
    filter_users.sort(key=lambda x: x['age'])
elif vbr_sort == '2':
    crit = input("Введите критерии поиска (первая буква логина): ")
    filter_users = [user for user in users if user['login'].lower().startswith(crit.lower())]
    filter_users.sort(key=lambda x: x['login'])
elif vbr_sort == '3':
    crit = input("Введите критерии поиска (название группы): ")
    filter_users = [user for user in users if user['group'].lower().startswith(crit.lower())]
    filter_users.sort(key=lambda x: x['group'])
else:
    print("Такого типа сортировки пока не существует, но мы работаем над этим. Пожалуйста, введите 1, 2 или 3.")

print("\nРезультат")
if not filter_users:
    print("Пользователь по вашему запрсу не найден.")
else:
    for user in filter_users:
        age = user['age']
        
print(f"#пользователь: '{user['login']}' возраст: {age} , группа: \"{user['group']}\"")
