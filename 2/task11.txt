#  todo: Дан номер месяца (1 — январь, 2 — февраль, ...). Вывести название соответствующего
#  времени года ("зима", "весна" и т.д.).


mes = int(input("Введите номер месяца (1-12):"))

if mes in [1, 2, 12]:
 vr_goda = "зима"
elif mes in [3, 4, 5]:
 vr_goda = "весна"
elif mes in [6, 7, 8]:
 vr_goda = "лето"
elif mes in [9, 10, 11]:
 vr_goda = "осень"
else:
 vr_goda = "Ошибка! Введите номер месяца от 1 до 12"

print(vr_goda)
