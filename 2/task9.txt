# todo: Решить линейное уравнение A·x + B = 0, заданное своими коэффициентами A и B (коэффициент A не равен 0).
# Примечание: коэффициенты получаем через функцию input().

try:
    A = float(input("Для решения уравнения A·x+B=0, введите значение числа A: "))
    B = float(input("Теперь введите значение числа B: "))
    if A == 0:
        print("'A' не может быть равно 0")
    else:
        print('A:', A, '|', 'B:', B)
        x = -B / A
        print(f"x = {x}")
except ValueError:
    print("Ошибка: перезапустите программу и введите корректное число, а не текст!")
