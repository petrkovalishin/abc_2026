#todo: Даны три точки A , B , C на числовой оси. Найти длины отрезков AC и BC и их сумму.
# Примечание: все точки получаем через функцию input().

try:
    A = float(input("Введите значение точки A: "))
    B = float(input("Введите значение точки B: "))
    C = float(input("Введите значение точки C: "))
    
except ValueError:
    print("Ошибка: перезапустите программу и введите корректное число, а не текст!")
    exit()
print('A:', A, '|', 'B:', B, '|', 'C:', C)
print(f"Длина отрезка AC: {abs(A + C)}")
print(f"Длина отрезка BC: {abs(B + C)}")
print(f"Сумарная длина отрезков AC и BC: {abs((A  + C) + (B + C))}")
