#todo: Дана сторона квадрата a. Найти его площадь S = a²
# Примечание: сторону квадрата получаем через функцию input().


try:
    a = float(input("Для вычисления площади введите значение равное стороне квадрата: "))
except ValueError:
    print("Ошибка: перезапустите программу и введите корректное число, а не текст!")
    exit()
print('Сторона квадрата:',a)
print(f"Площадь квадрата ровна: {a ** 2}")
