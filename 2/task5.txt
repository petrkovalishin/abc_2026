#todo: Написать программу, которая считывает два числа и выводит их сумму, разность, частное, произведение,
# результат целочисленного деления, результат деления с остатком, результат возведения в степень.


try:
    x = float(input("Введите первое число: "))  # Используем float, чтобы можно было вводить дробные значения
    y = float(input("Введите второе число: "))
except ValueError:
    print("Ошибка: перезапустите программу и введите корректное число, а не текст!")
    exit()
print('Первое число:',x, '|', 'Второе число:', y)
print(f"Сумма: {x + y}")
print(f"Разность: {x - y}")
print(f"Частное: {x / y}")
print(f"Произведение: {x * y}")
print(f"Целочисленное деление: {x // y}")
print(f"Остаток от деления: {x % y}")
print(f"Возведение в степень: {x ** y}")
