Ранее написанный логгер:

python
import logging
import sys


def logger(name: str = "app", level: int = logging.INFO) -> logging.Logger:
    log = logging.getLogger(name)
    log.setLevel(level)

    if not log.handlers:
        handler = logging.StreamHandler(sys.stdout)
        formatter = logging.Formatter(
            "%(asctime)s | %(name)s | %(levelname)s | %(message)s",
            datefmt="%Y-%m-%d %H:%M:%S",
        )
        handler.setFormatter(formatter)
        log.addHandler(handler)

    return log

Шифр Цезаря:
def caesar_cipher(text: str, shift: int = 3, decrypt: bool = False) -> str:
    if decrypt:
        shift = -shift

    result = []
    for ch in text:
        if ch.isalpha():
            base = ord('A') if ch.isupper() else ord('a')
            if 'А' <= ch <= 'я' or ch in 'Ёё':
                if ch.isupper():
                    base = ord('А')
                    alphabet_size = 32  # без Ё
                else:
                    base = ord('а')
                    alphabet_size = 32
                
                offset = ord(ch) - base
                if offset < 0 or offset >= alphabet_size:
                    result.append(ch)
                    continue
                new_offset = (offset + shift) % alphabet_size
                result.append(chr(base + new_offset))
            else:
                new_offset = (ord(ch) - base + shift) % 26
                result.append(chr(base + new_offset))
        else:
            result.append(ch)

    return "".join(result)

from helpers import logger, caesar_cipher

# --- Проверка логгера ---

# --- Проверка шифра Цезаря ---
original = "Hello, World!"
encrypted = caesar_cipher(original, shift=3)
decrypted = caesar_cipher(encrypted, shift=3, decrypt=True)

print(f"Оригинал:   {original}")
print(f"Зашифровано: {encrypted}")
print(f"Расшифровано: {decrypted}")

assert decrypted == original, "Ошибка шифрования!"
log.info("Проверка шифра Цезаря прошла успешно")
